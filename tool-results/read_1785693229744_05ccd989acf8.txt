   261→   261→/** Simple markdown-to-JSX renderer for AI responses */
   262→   262→function renderMarkdown(text: string) {
   263→   263→  const lines = text.split('\n');
   264→   264→  const elements: React.ReactNode[] = [];
   265→   265→  let key = 0;
   266→   266→
   267→   267→  for (const line of lines) {
   268→   268→    const trimmed = line.trim();
   269→   269→
   270→   270→    // Empty line → paragraph break
   271→   271→    if (!trimmed) {
   272→   272→      elements.push(<div key={key++} className="h-2" />);
   273→   273→      continue;
   274→   274→    }
   275→   275→
   276→   276→    // Bullet point (•)
   277→   277→    if (trimmed.startsWith('•') || trimmed.startsWith('- ')) {
   278→   278→      const content = trimmed.startsWith('•') ? trimmed.slice(1).trim() : trimmed.slice(2).trim();
   279→   279→      elements.push(
   280→   280→        <div key={key++} className="flex items-start gap-2 ml-1">
   281→   281→          <span className="w-1.5 h-1.5 rounded-full bg-primary mt-1.5 shrink-0" />
   282→   282→          <span>{renderInlineMarkdown(content)}</span>
   283→   283→        </div>
   284→   284→      );
   285→   285→      continue;
   286→   286→    }
   287→   287→
   288→   288→    // Numbered list (e.g. "1.")
   289→   289→    const numMatch = trimmed.match(/^(\d+)\.\s(.+)/);
   290→   290→    if (numMatch) {
   291→   291→      elements.push(
   292→   292→        <div key={key++} className="flex items-start gap-2 ml-1">
   293→   293→          <span className="text-primary font-medium text-xs mt-0.5 shrink-0 w-4 text-right">{numMatch[1]}.</span>
   294→   294→          <span>{renderInlineMarkdown(numMatch[2])}</span>
   295→   295→        </div>
   296→   296→      );
   297→   297→      continue;
   298→   298→    }
   299→   299→
   300→   300→    // Regular line
   301→   301→    elements.push(
   302→   302→      <p key={key++} className="leading-relaxed">
   303→   303→        {renderInlineMarkdown(trimmed)}
   304→   304→      </p>
   305→   305→    );
   306→   306→  }
   307→   307→
   308→   308→  return elements;
   309→   309→}

   311→   311→/** Render **bold** inline */
   312→   312→function renderInlineMarkdown(text: string): React.ReactNode {
   313→   313→  const parts = text.split(/(\*\*[^*]+\*\*)/g);
   314→   314→  return parts.map((part, i) => {
   315→   315→    if (part.startsWith('**') && part.endsWith('**')) {
   316→   316→      return (
   317→   317→        <strong key={i} className="font-semibold text-foreground">
   318→   318→          {part.slice(2, -2)}
   319→   319→        </strong>
   320→   320→      );
   321→   321→    }
   322→   322→    return part;
   323→   323→  });
   324→   324→}

   104→function OwnerDashboard() {
   105→  const ownerView = useAppStore((s) => s.ownerView);
   106→  const [isLoading, setIsLoading] = useState(false);
   107→  const [currentView, setCurrentView] = useState(ownerView);
   108→
   109→  useEffect(() => {
   110→    if (ownerView !== currentView) {
   111→      const startTimer = requestAnimationFrame(() => {
   112→        setIsLoading(true);
   113→      });
   114→      const timer = setTimeout(() => {
   115→        setCurrentView(ownerView);
   116→        setIsLoading(false);
   117→      }, 200);
   118→      return () => {
   119→        cancelAnimationFrame(startTimer);
   120→        clearTimeout(timer);
   121→      };
   122→    }
   123→  }, [ownerView]);
   124→
   125→  const renderView = useCallback(() => {
   126→    switch (currentView) {
   127→      case 'dashboard':
   128→        return <DashboardView />;
   129→      case 'today':
   130→        return <TodayView />;
   131→      case 'ai':
   132→        return <AIAssistant />;
   133→      case 'sales':
   134→        return <SalesView />;
   135→      case 'clients':
   136→        return <ClientsView />;
   137→      case 'promotions':
   138→        return <PromotionsView />;
   139→      case 'finance':
   140→        return <FinanceDashboard />;
   141→      case 'analytics':
   142→        return <AnalyticsView />;
   143→      case 'loyalty':
   144→        return <LoyaltyProgram />;
   145→      case 'goals':
   146→        return <GoalsView />;
   147→      case 'settings':
   148→        return <SettingsView />;
   149→      case 'reviews':
   150→        return <ReviewsManager />;
   151→      case 'team':
   152→        return <StaffOverview />;
   153→      case 'smm':
   154→        return <AIContent />;
   155→      default:
   156→        return <DashboardView />;
   157→    }
   158→  }, [currentView]);
   159→
   160→  return (
   161→    <div className="flex-1 overflow-hidden">
   162→      <ScrollArea className="h-full">
   163→        <AnimatePresence mode="wait">
   164→          {isLoading ? (
   165→            <motion.div
   166→              key="loading"
   167→              initial={{ opacity: 0 }}
   168→              animate={{ opacity: 1 }}
   169→              exit={{ opacity: 0 }}
   170→              transition={{ duration: 0.15 }}
   171→              className="p-4 md:p-6 lg:p-8 space-y-6"
   172→            >
   173→              <div className="grid grid-cols-2 lg:grid-cols-4 gap-4">
   174→                {Array.from({ length: 4 }).map((_, i) => (
   175→                  <Skeleton key={i} className="h-24 rounded-xl" />
   176→                ))}
   177→              </div>
   178→              <Skeleton className="h-48 rounded-xl" />
   179→              <div className="space-y-3">
   180→                {Array.from({ length: 3 }).map((_, i) => (
   181→                  <Skeleton key={i} className="h-16 rounded-xl" />
   182→                ))}
   183→              </div>
   184→            </motion.div>
   185→          ) : (
   186→            <motion.div
   187→              key={currentView}
   188→              initial={{ opacity: 0, y: 8 }}
   189→              animate={{ opacity: 1, y: 0 }}
   190→              exit={{ opacity: 0, y: -8 }}
   191→              transition={{ duration: 0.2, ease: 'easeInOut' }}
   192→              className="p-4 md:p-6 lg:p-8"
   193→            >
   194→              {renderView()}
   195→            </motion.div>
   196→          )}
   197→        </AnimatePresence>
   198→      </ScrollArea>
   199→    </div>
   200→  );
   201→}

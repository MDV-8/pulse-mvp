   158→   158→interface AdminTool {
   159→   159→  id: string;

   167→   167→  const [tools, setTools] = useState<AdminTool[]>(mockAdminTools);
   168→   168→  const [editingId, setEditingId] = useState<string | null>(null);

   172→   172→  const toggleTool = (id: string) => {
   173→   173→    setTools((prev) =>
   174→   174→      prev.map((t) => (t.id === id ? { ...t, enabled: !t.enabled } : t))
   175→   175→    );
   176→   176→  };

   179→   179→    setEditingId(tool.id);

   181→   181→    setEditDesc(tool.description);
   182→   182→  };

   185→   185→    if (!editingId) return;

   187→   187→      prev.map((t) =>
   188→   188→        t.id === editingId ? { ...t, name: editName, description: editDesc } : t

   190→   190→    );

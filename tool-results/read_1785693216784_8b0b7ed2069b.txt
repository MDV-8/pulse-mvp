   253→/** Variable typing delay based on response length */
   254→function getTypingDelay(response: string): number {
   255→  const len = response.length;
   256→  if (len < 100) return 800 + Math.random() * 400;  // 0.8-1.2s
   257→  if (len < 300) return 1200 + Math.random() * 600; // 1.2-1.8s
   258→  return 1800 + Math.random() * 700;                 // 1.8-2.5s
   259→}

   261→/** Simple markdown-to-JSX renderer for AI responses */
   262→function renderMarkdown(text: string) {
   263→  const lines = text.split('\n');
   264→  const elements: React.ReactNode[] = [];
   265→  let key = 0;
   266→
   267→  for (const line of lines) {
   268→    const trimmed = line.trim();
   269→
   270→    // Empty line → paragraph break
   271→    if (!trimmed) {
   272→      elements.push(<div key={key++} className="h-2" />);
   273→      continue;
   274→    }
   275→
   276→    // Bullet point (•)
   277→    if (trimmed.startsWith('•') || trimmed.startsWith('- ')) {
   278→      const content = trimmed.startsWith('•') ? trimmed.slice(1).trim() : trimmed.slice(2).trim();
   279→      elements.push(
   280→        <div key={key++} className="flex items-start gap-2 ml-1">
   281→          <span className="w-1.5 h-1.5 rounded-full bg-primary mt-1.5 shrink-0" />
   282→          <span>{renderInlineMarkdown(content)}</span>
   283→        </div>
   284→      );
   285→      continue;
   286→    }
   287→
   288→    // Numbered list (e.g. "1.")
   289→    const numMatch = trimmed.match(/^(\d+)\.\s(.+)/);
   290→    if (numMatch) {
   291→      elements.push(
   292→        <div key={key++} className="flex items-start gap-2 ml-1">
   293→          <span className="text-primary font-medium text-xs mt-0.5 shrink-0 w-4 text-right">{numMatch[1]}.</span>
   294→          <span>{renderInlineMarkdown(numMatch[2])}</span>
   295→        </div>
   296→      );
   297→      continue;
   298→    }
   299→
   300→    // Regular line
   301→    elements.push(
   302→      <p key={key++} className="leading-relaxed">
   303→        {renderInlineMarkdown(trimmed)}
   304→      </p>
   305→    );
   306→  }
   307→
   308→  return elements;
   309→}

   311→/** Render **bold** inline */
   312→function renderInlineMarkdown(text: string): React.ReactNode {
   313→  const parts = text.split(/(\*\*[^*]+\*\*)/g);
   314→  return parts.map((part, i) => {
   315→    if (part.startsWith('**') && part.endsWith('**')) {
   316→      return (
   317→        <strong key={i} className="font-semibold text-foreground">
   318→          {part.slice(2, -2)}
   319→        </strong>
   320→      );
   321→    }
   322→    return part;
   323→  });
   324→}

   428→            ? 'bg-primary text-primary-foreground rounded-br-sm'
   429→            : 'bg-card border border-border rounded-bl-sm'

   458→export function AIAssistant() {
   459→  const { aiChatMessages, addAIChatMessage, clearAIChat } = useAppStore();
   460→  const [inputValue, setInputValue] = useState('');
   461→  const [isTyping, setIsTyping] = useState(false);
   462→  const [aiMode, setAiMode] = useState<AIMode>('neural');
   463→  const scrollRef = useRef<HTMLDivElement>(null);
   464→  const inputRef = useRef<HTMLInputElement>(null);
   465→
   466→  useEffect(() => {
   467→    if (scrollRef.current) {
   468→      const viewport = scrollRef.current.querySelector('[data-radix-scroll-area-viewport]');
   469→      if (viewport) {
   470→        viewport.scrollTop = viewport.scrollHeight;
   471→      }
   472→    }
   473→  }, [aiChatMessages, isTyping]);
   474→
   475→  const handleSend = async (text?: string) => {
   476→    const messageText = text || inputValue.trim();
   477→    if (!messageText || isTyping) return;
   478→
   479→    addAIChatMessage({ role: 'user', content: messageText });
   480→    setInputValue('');
   481→    setIsTyping(true);
   482→
   483→    if (aiMode === 'neural') {
   484→      try {
   485→        const history = aiChatMessages.map((m) => ({
   486→          role: m.role,
   487→          content: m.content,
   488→        }));
   489→
   490→        const res = await fetch('/api/ai-chat', {
   491→          method: 'POST',
   492→          headers: { 'Content-Type': 'application/json' },
   493→          body: JSON.stringify({ messages: history }),
   494→        });
   495→
   496→        const data = await res.json();
   497→
   498→        if (data.success && data.response) {
   499→          setIsTyping(false);
   500→          addAIChatMessage({ role: 'assistant', content: data.response });
   501→          return;
   502→        }
   503→        // If API returns unsuccessful, fall through to keyword matching
   504→        console.warn('AI API returned unsuccessful, falling back to keywords');
   505→      } catch (err) {
   506→        // Network error or other issue — fall back to keyword matching
   507→        console.warn('AI API failed, falling back to keywords:', err);
   508→      }
   509→    }
   510→
   511→    // Keyword matching (default mode or fallback)
   512→    const response = getAIResponse(messageText);
   513→    const delay = getTypingDelay(response);
   514→
   515→    setTimeout(() => {
   516→      setIsTyping(false);
   517→      addAIChatMessage({ role: 'assistant', content: response });
   518→    }, delay);
   519→  };
   520→
   521→  const handleKeyDown = (e: React.KeyboardEvent<HTMLInputElement>) => {
   522→    if (e.key === 'Enter' && !e.shiftKey) {
   523→      e.preventDefault();
   524→      handleSend();
   525→    }
   526→  };
   527→
   528→  const showWelcome = aiChatMessages.length === 0 && !isTyping;
   529→
   530→  const welcomeMessage = useMemo(
   531→    () =>
   532→      'Привет! Я AI-ассистент PULSE 💜\n\nЯ анализирую данные вашего бизнеса "Coffee & Co" и помогаю принимать решения.\n\nЧто вас интересует прямо сейчас?',
   533→    []
   534→  );
   535→
   536→  return (
   537→    <div className="flex h-full flex-col">
   538→      {/* Header */}
   539→      <div className="flex items-center justify-between border-b border-border px-4 sm:px-6 py-4">
   540→        <div className="flex items-center gap-3">
   541→          <div className="flex h-9 w-9 items-center justify-center rounded-xl bg-primary/15">
   542→            <Brain className="h-5 w-5 text-primary" />
   543→          </div>
   544→          <div>
   545→            <h2 className="text-base font-semibold">AI Ассистент</h2>
   546→            <p className="text-xs text-muted-foreground">
   547→              Получает данные вашего бизнеса
   548→            </p>
   549→          </div>
   550→        </div>
   551→        <div className="flex items-center gap-2">
   552→          {/* AI Mode Toggle */}
   553→          <div className="flex items-center gap-1.5 rounded-lg bg-card border border-border p-0.5">
   554→            <button
   555→              onClick={() => setAiMode('keywords')}
   556→              className={cn(
   557→                'px-2.5 py-1 rounded-md text-xs font-medium transition-all',
   558→                aiMode === 'keywords'
   559→                  ? 'bg-primary text-primary-foreground shadow-sm'
   560→                  : 'text-muted-foreground hover:text-foreground'
   561→              )}
   562→            >
   563→              Ключевые слова
   564→            </button>
   565→            <button
   566→              onClick={() => setAiMode('neural')}
   567→              className={cn(
   568→                'px-2.5 py-1 rounded-md text-xs font-medium transition-all flex items-center gap-1.5',
   569→                aiMode === 'neural'
   570→                  ? 'bg-primary text-primary-foreground shadow-sm'
   571→                  : 'text-muted-foreground hover:text-foreground'
   572→              )}
   573→            >
   574→              <Zap className="h-3 w-3" />
   575→              Нейросеть
   576→              <span className={cn(
   577→                'text-[9px] font-bold px-1 py-0.5 rounded leading-none',
   578→                aiMode === 'neural'
   579→                  ? 'bg-primary-foreground/20 text-primary-foreground'
   580→                  : 'bg-primary/15 text-primary'
   581→              )}>
   582→                BETA
   583→              </span>
   584→            </button>
   585→          </div>
   586→          {aiChatMessages.length > 0 && (
   587→            <Button
   588→              variant="ghost"
   589→              size="sm"
   590→              className="text-xs text-muted-foreground hover:text-foreground"
   591→              onClick={clearAIChat}
   592→            >
   593→              Очистить
   594→            </Button>
   595→          )}
   596→        </div>
   597→      </div>

   599→      {/* Messages Area */}

   601→        <div className="space-y-4">
   602→          {/* Welcome Message */}

   604→            <motion.div
   605→              initial={{ opacity: 0, y: 12 }}
   606→              animate={{ opacity: 1, y: 0 }}
   607→              transition={{ duration: 0.4 }}
   608→              className="flex items-start gap-3"
   609→            >
   610→              <div className="flex h-8 w-8 shrink-0 items-center justify-center rounded-full bg-primary/20">
   611→                <Sparkles className="h-4 w-4 text-primary" />
   612→              </div>
   613→              <div className="rounded-2xl rounded-bl-sm bg-card border border-border px-4 py-3 text-sm leading-relaxed max-w-[85%]">
   614→                {renderMarkdown(welcomeMessage)}
   615→              </div>
   616→            </motion.div>
   617→          )}
   618→
   619→          {/* Chat Messages */}
   620→          {aiChatMessages.map((message) => (
   621→            <ChatMessage key={message.id} message={message} />
   622→          ))}
   623→
   624→          {/* Typing Indicator */}
   625→          <AnimatePresence>
   626→            {isTyping && (aiMode === 'neural' ? <NeuralTypingIndicator /> : <TypingIndicator />)}

   630→
   631→      {/* Suggested Questions */}
   632→      {showWelcome && (
   633→        <div className="border-t border-border px-4 sm:px-6 py-3">

   635→            {SUGGESTED_QUESTIONS.map((q) => (
   636→              <motion.button
   637→                key={q}
   638→                whileHover={{ scale: 1.02 }}
   639→                whileTap={{ scale: 0.98 }}
   640→                onClick={() => handleSend(q)}
   641→                className="rounded-full border border-border bg-card px-3 py-1.5 text-xs text-muted-foreground hover:border-primary/40 hover:text-foreground transition-colors"
   642→              >
   643→                {q}
   644→              </motion.button>
   645→            ))}
   646→          </div>
   647→        </div>
   648→      )}
   649→
   650→      {/* Disclaimer for Neural mode */}
   651→      {aiMode === 'neural' && (
   652→        <div className="px-4 sm:px-6">
   653→          <p className="text-[10px] text-muted-foreground/50 text-center">
   654→            AI ответы могут быть неточными. Демо режим.
   655→          </p>
   656→        </div>
   657→      )}
   658→
   659→      <Separator />
   660→
   661→      {/* Input Area */}
   662→      <div className="px-4 sm:px-6 py-3">
   663→        <div className="flex items-center gap-2">
   664→          <Input
   665→            ref={inputRef}
   666→            value={inputValue}
   667→            onChange={(e) => setInputValue(e.target.value)}
   668→            onKeyDown={handleKeyDown}
   669→            placeholder="Спросите что-нибудь..."
   670→            className="flex-1 bg-card border-border h-10 text-sm"
   671→            disabled={isTyping}
   672→          />
   673→          <Button
   674→            size="icon"
   675→            className="h-10 w-10 shrink-0 bg-primary hover:bg-primary/90 text-primary-foreground"
   676→            onClick={() => handleSend()}
   677→            disabled={!inputValue.trim() || isTyping}
   678→          >
   679→            <Send className="h-4 w-4" />
   680→          </Button>
   681→        </div>
   682→      </div>
   683→    </div>
   684→  );

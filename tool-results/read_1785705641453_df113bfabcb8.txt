   158→   158→// ---- Tools Tab ----
   159→   159→interface AdminTool {
   160→   160→  id: string;
   161→   161→  name: string;
   162→   162→  description: string;
   163→   163→  icon: string;
   164→   164→  enabled: boolean;
   165→   165→}
   166→   166→
   167→   167→const TOOLS_STORAGE_KEY = 'pulse-admin-tools';
   168→   168→
   169→   169→function loadTools(): AdminTool[] {
   170→   170→  if (typeof window === 'undefined') return mockAdminTools;
   171→   171→  try {
   172→   172→    const raw = localStorage.getItem(TOOLS_STORAGE_KEY);
   173→   173→    return raw ? JSON.parse(raw) : mockAdminTools;
   174→   174→  } catch { return mockAdminTools; }
   175→   175→}
   176→   176→
   177→   177→function saveTools(tools: AdminTool[]) {
   178→   178→  localStorage.setItem(TOOLS_STORAGE_KEY, JSON.stringify(tools));
   179→   179→}
   180→   180→
   181→   181→function ToolsTab() {
   182→   182→  const [tools, setTools] = useState<AdminTool[]>(loadTools);
   183→   183→  const [editingId, setEditingId] = useState<string | null>(null);
   184→   184→  const [editName, setEditName] = useState('');
   185→   185→  const [editDesc, setEditDesc] = useState('');
   186→   186→
   187→   187→  const updateTools = (updater: (prev: AdminTool[]) => AdminTool[]) => {
   188→   188→    setTools((prev) => {
   189→   189→      const next = updater(prev);
   190→   190→      saveTools(next);
   191→   191→      return next;
   192→   192→    });
   193→   193→  };
   194→   194→
   195→   195→  const toggleTool = (id: string) => {
   196→   196→    updateTools((prev) =>
   197→   197→      prev.map((t) => (t.id === id ? { ...t, enabled: !t.enabled } : t))
   198→   198→    );
   199→   199→  };
   200→   200→
   201→   201→  const startEdit = (tool: AdminTool) => {
   202→   202→    setEditingId(tool.id);
   203→   203→    setEditName(tool.name);
   204→   204→    setEditDesc(tool.description);
   205→   205→  };
   206→   206→
   207→   207→  const saveEdit = () => {
   208→   208→    if (!editingId) return;
   209→   209→    updateTools((prev) =>
   210→   210→      prev.map((t) =>
   211→   211→        t.id === editingId ? { ...t, name: editName, description: editDesc } : t
   212→   212→      )
   213→   213→    );
   214→   214→    setEditingId(null);
   215→   215→    toast.success('Инструмент обновлён');
   216→   216→  };
   217→   217→
   218→   218→  const deleteTool = (id: string) => {
   219→   219→    updateTools((prev) => prev.filter((t) => t.id !== id));
   220→   220→    if (editingId === id) setEditingId(null);
   221→   221→    toast.success('Инструмент удалён');
   222→   222→  };
   223→   223→
   224→   224→  const addTool = () => {
   225→   225→    const newTool: AdminTool = {
   226→   226→      id: `at-${Date.now()}`,
   227→   227→      name: 'Новый инструмент',

   229→   229→      icon: 'Lightbulb',
   230→   230→      enabled: false,
   231→   231→    };
   232→   232→    updateTools((prev) => [...prev, newTool]);
   233→   233→    startEdit(newTool);

   235→   235→  };

   237→   237→  return (

   242→   242→          <Plus className="w-4 h-4 mr-1" />
   243→   243→          Добавить
   244→   244→        </Button>
   245→   245→      </div>
   246→   246→
   247→   247→      <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
   248→   248→        {tools.map((tool, i) => {
   249→   249→          const IconComponent = iconMap[tool.icon] || Lightbulb;
   250→   250→          const isEditing = editingId === tool.id;
   251→   251→          const colorClass = toolColors[tool.icon] || toolColors['Lightbulb'];
   252→   252→
   253→   253→          return (
   254→   254→            <motion.div
   255→   255→              key={tool.id}
   256→   256→              initial={{ opacity: 0, y: 10 }}
   257→   257→              animate={{ opacity: 1, y: 0 }}
   258→   258→              transition={{ delay: i * 0.03 }}
   259→   259→            >

   261→   261→                <CardContent className="p-4 space-y-3">
   262→   262→                  <div className="flex items-start justify-between">
   263→   263→                    <div className="flex items-start gap-3">

   270→   270→                          <Input

   272→   272→                            onChange={(e) => setEditName(e.target.value)}

   285→   285→                            {tool.description}

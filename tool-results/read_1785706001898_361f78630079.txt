   158→// ---- Tools Tab ----
   159→interface AdminTool {
   160→  id: string;
   161→  name: string;
   162→  description: string;
   163→  icon: string;
   164→  enabled: boolean;
   165→}
   166→
   167→const TOOLS_STORAGE_KEY = 'pulse-admin-tools';
   168→
   169→function loadTools(): AdminTool[] {
   170→  if (typeof window === 'undefined') return mockAdminTools;
   171→  try {
   172→    const raw = localStorage.getItem(TOOLS_STORAGE_KEY);
   173→    return raw ? JSON.parse(raw) : mockAdminTools;
   174→  } catch { return mockAdminTools; }
   175→}
   176→
   177→function saveTools(tools: AdminTool[]) {
   178→  localStorage.setItem(TOOLS_STORAGE_KEY, JSON.stringify(tools));
   179→}
   180→
   181→function ToolsTab() {
   182→  const [tools, setTools] = useState<AdminTool[]>(loadTools);
   183→  const [editingId, setEditingId] = useState<string | null>(null);
   184→  const [editName, setEditName] = useState('');
   185→  const [editDesc, setEditDesc] = useState('');
   186→
   187→  const updateTools = (updater: (prev: AdminTool[]) => AdminTool[]) => {
   188→    setTools((prev) => {
   189→      const next = updater(prev);
   190→      saveTools(next);
   191→      return next;
   192→    });
   193→  };
   194→
   195→  const toggleTool = (id: string) => {
   196→    updateTools((prev) =>
   197→      prev.map((t) => (t.id === id ? { ...t, enabled: !t.enabled } : t))
   198→    );
   199→  };
   200→
   201→  const startEdit = (tool: AdminTool) => {
   202→    setEditingId(tool.id);
   203→    setEditName(tool.name);
   204→    setEditDesc(tool.description);
   205→  };
   206→
   207→  const saveEdit = () => {
   208→    if (!editingId) return;
   209→    updateTools((prev) =>
   210→      prev.map((t) =>
   211→        t.id === editingId ? { ...t, name: editName, description: editDesc } : t
   212→      )
   213→    );
   214→    setEditingId(null);
   215→    toast.success('Инструмент обновлён');
   216→  };
   217→
   218→  const deleteTool = (id: string) => {
   219→    updateTools((prev) => prev.filter((t) => t.id !== id));
   220→    if (editingId === id) setEditingId(null);
   221→    toast.success('Инструмент удалён');
   222→  };
   223→
   224→  const addTool = () => {
   225→    const newTool: AdminTool = {
   226→      id: `at-${Date.now()}`,
   227→      name: 'Новый инструмент',
   228→      description: 'Описание инструмента',
   229→      icon: 'Lightbulb',
   230→      enabled: false,
   231→    };
   232→    updateTools((prev) => [...prev, newTool]);
   233→    startEdit(newTool);
   234→    toast.success('✅ Инструмент успешно опубликован');
   235→  };
   236→
   237→  return (
   238→    <div className="space-y-4">
   239→      <div className="flex items-center justify-between">
   240→        <h2 className="text-lg font-semibold">Инструменты</h2>
   241→        <Button size="sm" className="bg-purple-600 hover:bg-purple-700 text-white" onClick={addTool}>
   242→          <Plus className="w-4 h-4 mr-1" />
   243→          Добавить
   244→        </Button>
   245→      </div>
   246→
   247→      <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
   248→        {tools.map((tool, i) => {
   249→          const IconComponent = iconMap[tool.icon] || Lightbulb;
   250→          const isEditing = editingId === tool.id;
   251→          const colorClass = toolColors[tool.icon] || toolColors['Lightbulb'];
   252→
   253→          return (
   254→            <motion.div
   255→              key={tool.id}
   256→              initial={{ opacity: 0, y: 10 }}
   257→              animate={{ opacity: 1, y: 0 }}
   258→              transition={{ delay: i * 0.03 }}
   259→            >
   260→              <Card className={cn('p-0 transition-opacity card-hover-lift micro-interaction', !tool.enabled && 'opacity-60')}>
   261→                <CardContent className="p-4 space-y-3">
   262→                  <div className="flex items-start justify-between">
   263→                    <div className="flex items-start gap-3">
   264→                      {/* Circular colored icon container */}
   265→                      <div className={cn('w-10 h-10 rounded-full flex items-center justify-center shrink-0', colorClass)}>
   266→                        <IconComponent className="w-5 h-5" />
   267→                      </div>
   268→                      {isEditing ? (
   269→                        <div className="space-y-1 flex-1">
   270→                          <Input
   271→                            value={editName}
   272→                            onChange={(e) => setEditName(e.target.value)}
   273→                            className="h-7 text-sm"
   274→                          />
   275→                          <Input
   276→                            value={editDesc}
   277→                            onChange={(e) => setEditDesc(e.target.value)}
   278→                            className="h-7 text-sm"
   279→                          />
   280→                        </div>
   281→                      ) : (
   282→                        <div>
   283→                          <h3 className="font-semibold text-sm">{tool.name}</h3>
   284→                          <p className="text-xs text-muted-foreground mt-0.5">
   285→                            {tool.description}
   286→                          </p>
   287→                        </div>
   288→                      )}
   289→                    </div>
   290→                    <div className="flex items-center gap-1 shrink-0">
   291→                      {isEditing ? (
   292→                        <Button size="sm" variant="ghost" onClick={saveEdit} className="text-green-400 hover:text-green-300">
   293→                          <Check className="w-4 h-4" />
   294→                        </Button>
   295→                      ) : (
   296→                        <Button size="sm" variant="ghost" onClick={() => startEdit(tool)} className="text-muted-foreground hover:text-foreground">
   297→                          <Pencil className="w-4 h-4" />
   298→                        </Button>
   299→                      )}
   300→                      <Button size="sm" variant="ghost" onClick={() => deleteTool(tool.id)} className="text-muted-foreground hover:text-red-400">
   301→                        <Trash2 className="w-4 h-4" />
   302→                      </Button>
   303→                    </div>
   304→                  </div>
   305→                  <div className="flex items-center justify-between">
   306→                    <span className={cn(
   307→                      'text-xs font-medium',
   308→                      tool.enabled ? 'text-green-400' : 'text-red-400/70'
   309→                    )}>
   310→                      {tool.enabled ? '● Включён' : '○ Выключен'}
   311→                    </span>
   312→                    <Switch checked={tool.enabled} onCheckedChange={() => toggleTool(tool.id)} />
   313→                  </div>
   314→                </CardContent>
   315→              </Card>
   316→            </motion.div>
   317→          );
   318→        })}
   319→      </div>
   320→    </div>
   321→  );
   322→}

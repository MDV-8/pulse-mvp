   162→  description: string;

   167→function ToolsTab() {

   170→  const [editName, setEditName] = useState('');

   172→
   173→  const toggleTool = (id: string) => {
   174→    setTools((prev) =>
   175→      prev.map((t) => (t.id === id ? { ...t, enabled: !t.enabled } : t))
   176→    );
   177→  };
   178→

   180→    setEditingId(tool.id);
   181→    setEditName(tool.name);
   182→    setEditDesc(tool.description);
   183→  };

   185→  const saveEdit = () => {
   186→    if (!editingId) return;
   187→    setTools((prev) =>
   188→      prev.map((t) =>
   189→        t.id === editingId ? { ...t, name: editName, description: editDesc } : t
   190→      )
   191→    );
   192→    setEditingId(null);
   193→  };
   194→
   195→  const addTool = () => {
   196→    const newTool: AdminTool = {
   197→      id: `at-${Date.now()}`,
   198→      name: 'Новый инструмент',
   199→      description: 'Описание инструмента',

   202→    };

   205→  };

   207→  return (
   208→    <div className="space-y-4">
   209→      <div className="flex items-center justify-between">
   210→        <h2 className="text-lg font-semibold">Инструменты</h2>

   214→        </Button>
   215→      </div>
   216→
   217→      <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
   218→        {tools.map((tool, i) => {
   219→          const IconComponent = iconMap[tool.icon] || Lightbulb;
   220→          const isEditing = editingId === tool.id;
   221→          const colorClass = toolColors[tool.icon] || toolColors['Lightbulb'];
   222→
   223→          return (
   224→            <motion.div
   225→              key={tool.id}
   226→              initial={{ opacity: 0, y: 10 }}
   227→              animate={{ opacity: 1, y: 0 }}
   228→              transition={{ delay: i * 0.03 }}
   229→            >
   230→              <Card className={cn('p-0 transition-opacity card-hover-lift micro-interaction', !tool.enabled && 'opacity-60')}>
   231→                <CardContent className="p-4 space-y-3">
   232→                  <div className="flex items-start justify-between">
   233→                    <div className="flex items-start gap-3">

   239→                        <div className="space-y-1 flex-1">
   240→                          <Input

   243→                            className="h-7 text-sm"
   244→                          />
   245→                          <Input
   246→                            value={editDesc}
   247→                            onChange={(e) => setEditDesc(e.target.value)}
   248→                            className="h-7 text-sm"
   249→                          />
   250→                        </div>
   251→                      ) : (
   252→                        <div>
   253→                          <h3 className="font-semibold text-sm">{tool.name}</h3>
   254→                          <p className="text-xs text-muted-foreground mt-0.5">
   255→                            {tool.description}

   262→                        <Button size="sm" variant="ghost" onClick={saveEdit} className="text-green-400 hover:text-green-300">
   263→                          <Check className="w-4 h-4" />
   264→                        </Button>
   265→                      ) : (
   266→                        <Button size="sm" variant="ghost" onClick={() => startEdit(tool)} className="text-muted-foreground hover:text-foreground">
   267→                          <Pencil className="w-4 h-4" />

   269→                      )}
   270→                    </div>
   271→                  </div>

    93→    93→function OwnerDashboard() {
    94→    94→  const ownerView = useAppStore((s) => s.ownerView);
    95→    95→  const [isLoading, setIsLoading] = useState(false);
    96→    96→  const [currentView, setCurrentView] = useState(ownerView);
    97→    97→
    98→    98→  useEffect(() => {
    99→    99→    if (ownerView !== currentView) {
   100→   100→      const startTimer = requestAnimationFrame(() => {
   101→   101→        setIsLoading(true);
   102→   102→      });
   103→   103→      const timer = setTimeout(() => {
   104→   104→        setCurrentView(ownerView);
   105→   105→        setIsLoading(false);
   106→   106→      }, 200);
   107→   107→      return () => {
   108→   108→        cancelAnimationFrame(startTimer);
   109→   109→        clearTimeout(timer);
   110→   110→      };
   111→   111→    }
   112→   112→  }, [ownerView]);
   113→   113→
   114→   114→  const renderView = useCallback(() => {
   115→   115→    switch (currentView) {
   116→   116→      case 'dashboard':
   117→   117→        return <DashboardView />;
   118→   118→      case 'today':
   119→   119→        return <TodayView />;
   120→   120→      case 'ai':
   121→   121→        return <AIAssistant />;
   122→   122→      case 'sales':
   123→   123→        return <SalesView />;
   124→   124→      case 'clients':
   125→   125→        return <ClientsView />;
   126→   126→      case 'promotions':
   127→   127→        return <PromotionsView />;
   128→   128→      case 'finance':
   129→   129→        return <FinanceDashboard />;
   130→   130→      case 'analytics':
   131→   131→        return <AnalyticsView />;
   132→   132→      case 'loyalty':
   133→   133→        return <LoyaltyProgram />;
   134→   134→      case 'goals':
   135→   135→        return <GoalsView />;
   136→   136→      case 'settings':
   137→   137→        return <SettingsView />;
   138→   138→      default:
   139→   139→        return <DashboardView />;
   140→   140→    }
   141→   141→  }, [currentView]);
   142→   142→
   143→   143→  return (
   144→   144→    <div className="flex-1 overflow-hidden">
   145→   145→      <ScrollArea className="h-full">
   146→   146→        <AnimatePresence mode="wait">
   147→   147→          {isLoading ? (
   148→   148→            <motion.div
   149→   149→              key="loading"
   150→   150→              initial={{ opacity: 0 }}
   151→   151→              animate={{ opacity: 1 }}
   152→   152→              exit={{ opacity: 0 }}
   153→   153→              transition={{ duration: 0.15 }}
   154→   154→              className="p-4 md:p-6 lg:p-8 space-y-6"
   155→   155→            >
   156→   156→              <div className="grid grid-cols-2 lg:grid-cols-4 gap-4">
   157→   157→                {Array.from({ length: 4 }).map((_, i) => (
   158→   158→                  <Skeleton key={i} className="h-24 rounded-xl" />
   159→   159→                ))}
   160→   160→              </div>
   161→   161→              <Skeleton className="h-48 rounded-xl" />
   162→   162→              <div className="space-y-3">
   163→   163→                {Array.from({ length: 3 }).map((_, i) => (
   164→   164→                  <Skeleton key={i} className="h-16 rounded-xl" />
   165→   165→                ))}
   166→   166→              </div>
   167→   167→            </motion.div>
   168→   168→          ) : (
   169→   169→            <motion.div
   170→   170→              key={currentView}
   171→   171→              initial={{ opacity: 0, y: 8 }}
   172→   172→              animate={{ opacity: 1, y: 0 }}
   173→   173→              exit={{ opacity: 0, y: -8 }}
   174→   174→              transition={{ duration: 0.2, ease: 'easeInOut' }}
   175→   175→              className="p-4 md:p-6 lg:p-8"
   176→   176→            >
   177→   177→              {renderView()}
   178→   178→            </motion.div>
   179→   179→          )}
   180→   180→        </AnimatePresence>
   181→   181→      </ScrollArea>
   182→   182→    </div>
   183→   183→  );
   184→   184→}

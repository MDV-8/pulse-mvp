   109→// ============================================================
   110→function OwnerDashboard() {
   111→  const ownerView = useAppStore((s) => s.ownerView);
   112→  const [isLoading, setIsLoading] = useState(false);
   113→  const [currentView, setCurrentView] = useState(ownerView);
   114→
   115→  useEffect(() => {
   116→    if (ownerView !== currentView) {
   117→      const startTimer = requestAnimationFrame(() => {
   118→        setIsLoading(true);
   119→      });
   120→      const timer = setTimeout(() => {
   121→        setCurrentView(ownerView);
   122→        setIsLoading(false);
   123→      }, 200);
   124→      return () => {
   125→        cancelAnimationFrame(startTimer);
   126→        clearTimeout(timer);
   127→      };
   128→    }
   129→  }, [ownerView]);
   130→
   131→  const renderView = useCallback(() => {
   132→    switch (currentView) {
   133→      case 'dashboard':
   134→        return <DashboardView />;
   135→      case 'today':
   136→        return <TodayView />;
   137→      case 'ai':
   138→        return <AIAssistant />;
   139→      case 'sales':
   140→        return <SalesView />;
   141→      case 'clients':
   142→        return <ClientsView />;
   143→      case 'promotions':
   144→        return <PromotionsView />;
   145→      case 'finance':
   146→        return <FinanceDashboard />;
   147→      case 'analytics':
   148→        return <AnalyticsView />;
   149→      case 'loyalty':
   150→        return <LoyaltyProgram />;
   151→      case 'goals':
   152→        return <GoalsView />;
   153→      case 'settings':
   154→        return <SettingsView />;
   155→      case 'reviews':
   156→        return <ReviewsManager />;
   157→      case 'team':
   158→        return <StaffOverview />;
   159→      case 'smm':
   160→        return <AIContent />;
   161→      case 'inventory':
   162→        return <InventoryList />;
   163→      default:
   164→        return <DashboardView />;
   165→    }
   166→  }, [currentView]);
   167→
   168→  return (
   169→    <div className="flex-1 overflow-hidden">
   170→      <ScrollArea className="h-full">
   171→        <AnimatePresence mode="wait">
   172→          {isLoading ? (
   173→            <motion.div
   174→              key="loading"
   175→              initial={{ opacity: 0 }}
   176→              animate={{ opacity: 1 }}
   177→              exit={{ opacity: 0 }}
   178→              transition={{ duration: 0.15 }}
   179→              className="p-4 md:p-6 lg:p-8 space-y-6"
   180→            >
   181→              <div className="grid grid-cols-2 lg:grid-cols-4 gap-4">
   182→                {Array.from({ length: 4 }).map((_, i) => (
   183→                  <Skeleton key={i} className="h-24 rounded-xl" />
   184→                ))}
   185→              </div>
   186→              <Skeleton className="h-48 rounded-xl" />
   187→              <div className="space-y-3">
   188→                {Array.from({ length: 3 }).map((_, i) => (
   189→                  <Skeleton key={i} className="h-16 rounded-xl" />
   190→                ))}
   191→              </div>
   192→            </motion.div>
   193→          ) : (
   194→            <motion.div
   195→              key={currentView}
   196→              initial={{ opacity: 0, y: 8 }}
   197→              animate={{ opacity: 1, y: 0 }}
   198→              exit={{ opacity: 0, y: -8 }}
   199→              transition={{ duration: 0.2, ease: 'easeInOut' }}
   200→              className="p-4 md:p-6 lg:p-8"
   201→            >
   202→              {renderView()}
   203→            </motion.div>
   204→          )}
   205→        </AnimatePresence>
   206→      </ScrollArea>
   207→    </div>
   208→  );
   209→}

   166→function ToolsTab() {

   169→  const [editName, setEditName] = useState('');
   170→  const [editDesc, setEditDesc] = useState('');
   171→
   172→  const toggleTool = (id: string) => {
   173→    setTools((prev) =>
   174→      prev.map((t) => (t.id === id ? { ...t, enabled: !t.enabled } : t))
   175→    );
   176→  };
   177→
   178→  const startEdit = (tool: AdminTool) => {
   179→    setEditingId(tool.id);
   180→    setEditName(tool.name);
   181→    setEditDesc(tool.description);
   182→  };
   183→
   184→  const saveEdit = () => {
   185→    if (!editingId) return;
   186→    setTools((prev) =>
   187→      prev.map((t) =>
   188→        t.id === editingId ? { ...t, name: editName, description: editDesc } : t
   189→      )
   190→    );
   191→    setEditingId(null);
   192→  };
   193→
   194→  const addTool = () => {
   195→    const newTool: AdminTool = {
   196→      id: `at-${Date.now()}`,
   197→      name: 'Новый инструмент',
   198→      description: 'Описание инструмента',
   199→      icon: 'Lightbulb',
   200→      enabled: false,
   201→    };
   202→    setTools((prev) => [...prev, newTool]);

   204→  };
   205→
   206→  return (
   207→    <div className="space-y-4">
   208→      <div className="flex items-center justify-between">
   209→        <h2 className="text-lg font-semibold">Инструменты</h2>
   210→        <Button size="sm" className="bg-purple-600 hover:bg-purple-700 text-white" onClick={addTool}>
   211→          <Plus className="w-4 h-4 mr-1" />
   212→          Добавить
   213→        </Button>
   214→      </div>
   215→
   216→      <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
   217→        {tools.map((tool, i) => {
   218→          const IconComponent = iconMap[tool.icon] || Lightbulb;
   219→          const isEditing = editingId === tool.id;
   220→          const colorClass = toolColors[tool.icon] || toolColors['Lightbulb'];
   221→
   222→          return (
   223→            <motion.div
   224→              key={tool.id}
   225→              initial={{ opacity: 0, y: 10 }}
   226→              animate={{ opacity: 1, y: 0 }}
   227→              transition={{ delay: i * 0.03 }}
   228→            >
   229→              <Card className={cn('p-0 transition-opacity card-hover-lift micro-interaction', !tool.enabled && 'opacity-60')}>
   230→                <CardContent className="p-4 space-y-3">
   231→                  <div className="flex items-start justify-between">
   232→                    <div className="flex items-start gap-3">
   233→                      {/* Circular colored icon container */}
   234→                      <div className={cn('w-10 h-10 rounded-full flex items-center justify-center shrink-0', colorClass)}>
   235→                        <IconComponent className="w-5 h-5" />

   238→                        <div className="space-y-1 flex-1">
   239→                          <Input

   242→                            className="h-7 text-sm"
   243→                          />
   244→                          <Input
   245→                            value={editDesc}
   246→                            onChange={(e) => setEditDesc(e.target.value)}
   247→                            className="h-7 text-sm"
   248→                          />
   249→                        </div>
   250→                      ) : (
   251→                        <div>
   252→                          <h3 className="font-semibold text-sm">{tool.name}</h3>
   253→                          <p className="text-xs text-muted-foreground mt-0.5">
   254→                            {tool.description}
   255→                          </p>

   261→                        <Button size="sm" variant="ghost" onClick={saveEdit} className="text-green-400 hover:text-green-300">
   262→                          <Check className="w-4 h-4" />
   263→                        </Button>
   264→                      ) : (
   265→                        <Button size="sm" variant="ghost" onClick={() => startEdit(tool)} className="text-muted-foreground hover:text-foreground">
   266→                          <Pencil className="w-4 h-4" />

   268→                      )}
   269→                    </div>
   270→                  </div>

   284→        })}

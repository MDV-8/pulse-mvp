    93→function OwnerDashboard() {
    94→  const ownerView = useAppStore((s) => s.ownerView);
    95→  const [isLoading, setIsLoading] = useState(false);
    96→  const [currentView, setCurrentView] = useState(ownerView);
    97→
    98→  useEffect(() => {
    99→    if (ownerView !== currentView) {
   100→      const startTimer = requestAnimationFrame(() => {
   101→        setIsLoading(true);
   102→      });
   103→      const timer = setTimeout(() => {
   104→        setCurrentView(ownerView);
   105→        setIsLoading(false);
   106→      }, 200);
   107→      return () => {
   108→        cancelAnimationFrame(startTimer);
   109→        clearTimeout(timer);
   110→      };
   111→    }
   112→  }, [ownerView]);
   113→
   114→  const renderView = useCallback(() => {
   115→    switch (currentView) {
   116→      case 'dashboard':
   117→        return <DashboardView />;
   118→      case 'today':
   119→        return <TodayView />;
   120→      case 'ai':
   121→        return <AIAssistant />;
   122→      case 'sales':
   123→        return <SalesView />;
   124→      case 'clients':
   125→        return <ClientsView />;
   126→      case 'promotions':
   127→        return <PromotionsView />;
   128→      case 'finance':
   129→        return <FinanceDashboard />;
   130→      case 'analytics':
   131→        return <AnalyticsView />;
   132→      case 'loyalty':
   133→        return <LoyaltyProgram />;
   134→      case 'goals':
   135→        return <GoalsView />;
   136→      case 'settings':
   137→        return <SettingsView />;
   138→      default:
   139→        return <DashboardView />;
   140→    }
   141→  }, [currentView]);
   142→
   143→  return (
   144→    <div className="flex-1 overflow-hidden">
   145→      <ScrollArea className="h-full">
   146→        <AnimatePresence mode="wait">
   147→          {isLoading ? (
   148→            <motion.div
   149→              key="loading"
   150→              initial={{ opacity: 0 }}
   151→              animate={{ opacity: 1 }}
   152→              exit={{ opacity: 0 }}
   153→              transition={{ duration: 0.15 }}
   154→              className="p-4 md:p-6 lg:p-8 space-y-6"
   155→            >
   156→              <div className="grid grid-cols-2 lg:grid-cols-4 gap-4">
   157→                {Array.from({ length: 4 }).map((_, i) => (
   158→                  <Skeleton key={i} className="h-24 rounded-xl" />
   159→                ))}
   160→              </div>
   161→              <Skeleton className="h-48 rounded-xl" />
   162→              <div className="space-y-3">
   163→                {Array.from({ length: 3 }).map((_, i) => (
   164→                  <Skeleton key={i} className="h-16 rounded-xl" />
   165→                ))}
   166→              </div>
   167→            </motion.div>
   168→          ) : (
   169→            <motion.div
   170→              key={currentView}
   171→              initial={{ opacity: 0, y: 8 }}
   172→              animate={{ opacity: 1, y: 0 }}
   173→              exit={{ opacity: 0, y: -8 }}
   174→              transition={{ duration: 0.2, ease: 'easeInOut' }}
   175→              className="p-4 md:p-6 lg:p-8"
   176→            >
   177→              {renderView()}
   178→            </motion.div>
   179→          )}
   180→        </AnimatePresence>
   181→      </ScrollArea>
   182→    </div>
   183→  );
   184→}
